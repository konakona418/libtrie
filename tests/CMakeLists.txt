function(trie_add_test name)
    add_executable(${name} ${name}.cc)
    target_link_libraries(${name} PRIVATE trie::trie)
    target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_options(${name} PRIVATE -Wall -Wextra -fno-exceptions -fno-rtti)
    add_test(NAME ${name} COMMAND ${name})
endfunction()

# nghttp2 is optional; resolve it through pkg-config with a find_path/find_library
# fallback so HTTP/2 tests build on distros without a pkg-config file.
set(TRIE_NGHTTP2_TARGET "")
if(PkgConfig_FOUND)
    pkg_check_modules(NGHTTP2 QUIET IMPORTED_TARGET libnghttp2)
    if(NGHTTP2_FOUND)
        set(TRIE_NGHTTP2_TARGET PkgConfig::NGHTTP2)
    endif()
endif()
if(NOT TRIE_NGHTTP2_TARGET)
    find_path(NGHTTP2_INCLUDE_DIR nghttp2/nghttp2.h)
    find_library(NGHTTP2_LIBRARY NAMES nghttp2)
    if(NGHTTP2_INCLUDE_DIR AND NGHTTP2_LIBRARY)
        add_library(trie_nghttp2 INTERFACE)
        target_include_directories(trie_nghttp2 INTERFACE ${NGHTTP2_INCLUDE_DIR})
        target_link_libraries(trie_nghttp2 INTERFACE ${NGHTTP2_LIBRARY})
        set(TRIE_NGHTTP2_TARGET trie_nghttp2)
    endif()
endif()

endif()
